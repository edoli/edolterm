cmake_minimum_required(VERSION 3.21)
project(EdoltermExplorer LANGUAGES CXX)
add_library(EdoltermExplorer SHARED ExplorerCommand.cpp ExplorerCommand.def)
set_target_properties(EdoltermExplorer PROPERTIES OUTPUT_NAME "Edolterm.ExplorerCommand"
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_features(EdoltermExplorer PRIVATE cxx_std_17)
target_compile_options(EdoltermExplorer PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(EdoltermExplorer PRIVATE shell32 shlwapi ole32 uuid)
option(BUILD_TESTING "Build optional Explorer integration tests" OFF)
include(CTest)
if(BUILD_TESTING)
    add_executable(ExplorerCommandTests tests/ExplorerCommandTests.cpp)
    add_executable(TestLauncher tests/Launcher.cpp)
    set_target_properties(TestLauncher PROPERTIES OUTPUT_NAME "edolterm")
    foreach(target ExplorerCommandTests TestLauncher)
        target_compile_features(${target} PRIVATE cxx_std_17)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /EHsc)
        set_target_properties(${target} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
    target_link_libraries(ExplorerCommandTests PRIVATE shell32 ole32 uuid)
    add_test(NAME ExplorerCommand COMMAND ExplorerCommandTests "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
endif()
